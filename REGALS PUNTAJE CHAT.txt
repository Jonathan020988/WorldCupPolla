func calcularPuntos(prediction, match):
    if match.finalized == false: return 0
    puntosBase = (match.stage in GROUPS) ? basePoints : basePoints * 2

    if prediction.pred_home == match.home AND prediction.pred_away == match.away:
        total = puntosBase.exacto  # 10 o 20
        detalle = { exacto: total, resto: 0 }
    else:
        total = 0
        if sign(pred_home - pred_away) == sign(match_home - match_away):
            total += puntosBase.correcto  # 4 o 8
            correcto = puntosBase.correcto
        # comprobar goles exactos por equipo:
        goles = 0
        if prediction.pred_home == match.home: goles += puntosBase.goles  # 2 o 4
        if prediction.pred_away == match.away: goles += puntosBase.goles
        # comprobar diferencia (mutuamente excluyente con goles)
        diffPred = abs(pred_home - pred_away)
        diffReal = abs(match_home - match_away)
        diferencia = 0
        if goles == 0 and diffPred == diffReal:
            diferencia = puntosBase.difference  # 1 o 2

        total += correcto + max(goles, diferencia)
    return total
